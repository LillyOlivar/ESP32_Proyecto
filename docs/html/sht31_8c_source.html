33static uint8_t sht31_crc8(const uint8_t *data, int len)
34{
35 uint8_t crc = 0xFF;
36 for (int i = 0; i < len; i++) {
37 crc ^= data[i];
38 for (int b = 0; b < 8; b++) {
39 crc = (crc & 0x80) ? (uint8_t)((crc << 1) ^ 0x31) : (uint8_t)(crc << 1);
40 }
41 }
42 return crc;
43}

83esp_err_t sht31_read(float *temp_c, float *humidity_pct)
84{
85 if (temp_c == NULL || humidity_pct == NULL) {
86 return ESP_ERR_INVALID_ARG;
87 }
88 if (s_dev == NULL) {
89 return ESP_ERR_INVALID_STATE;
90 }
91
92 esp_err_t err = i2c_master_transmit(s_dev, SHT31_CMD_MEASURE_HIGH, sizeof(SHT31_CMD_MEASURE_HIGH),
93 SHT31_I2C_TIMEOUT_MS);
94 if (err != ESP_OK) {
95 return err;
96 }
97 vTaskDelay(pdMS_TO_TICKS(SHT31_MEASURE_DELAY_MS));
98
99 uint8_t buf[6]; /* T_MSB, T_LSB, T_CRC, RH_MSB, RH_LSB, RH_CRC */
100 err = i2c_master_receive(s_dev, buf, sizeof(buf), SHT31_I2C_TIMEOUT_MS);
101 if (err != ESP_OK) {
102 return err;
103 }
104 if (sht31_crc8(&buf[0], 2) != buf[2] || sht31_crc8(&buf[3], 2) != buf[5]) {
105 ESP_LOGW(TAG, "CRC invalido en la lectura");
106 return ESP_ERR_INVALID_CRC;
107 }
108
109 uint16_t raw_t = ((uint16_t)buf[0] << 8) | buf[1];
110 uint16_t raw_rh = ((uint16_t)buf[3] << 8) | buf[4];
111 *temp_c = -45.0f + 175.0f * ((float)raw_t / 65535.0f); /* formula de la hoja de datos */
112 *humidity_pct = 100.0f * ((float)raw_rh / 65535.0f);
113 return ESP_OK;
114}
