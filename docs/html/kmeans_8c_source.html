51esp_err_t kmeans_predict(float temp, float humidity, kmeans_result_t *result)
52{
53 if (result == NULL || isnan(temp) || isnan(humidity)) {
54 return ESP_ERR_INVALID_ARG;
55 }
56
57 result->cluster_id = -1;
58 result->cluster_label = "Desconocido";
59 result->min_distance = 1e9f;
60
61 /* 1. Normalizar para dar igual peso a Temp y Humedad */
62 float norm_feat[NUM_FEATURES];
63 norm_feat[0] = normalize(temp, TEMP_MIN, TEMP_MAX);
64 norm_feat[1] = normalize(humidity, HUM_MIN, HUM_MAX);
65
66 /* 2. Centroide mas cercano (distancia euclidiana al cuadrado, sin sqrtf en el bucle) */
67 for (int i = 0; i < K_CLUSTERS; i++) {
68 float d_temp = norm_feat[0] - clusters[i].centroid[0];
69 float d_hum = norm_feat[1] - clusters[i].centroid[1];
70 float dist_sq = (d_temp * d_temp) + (d_hum * d_hum);
71
72 if (dist_sq < result->min_distance) {
73 result->min_distance = dist_sq;
74 result->cluster_id = clusters[i].id;
75 result->cluster_label = clusters[i].label;
76 }
77 }
78
79 /* 3. Distancia euclidiana real como metrica final */
80 result->min_distance = sqrtf(result->min_distance);
81 return ESP_OK;
82}
