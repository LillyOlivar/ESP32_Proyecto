45esp_err_t ota_client_run(const ota_client_config_t *cfg)
46{
47 if (cfg == NULL || cfg->url == NULL || cfg->url[0] == '\0' ||
48 cfg->cert_pem == NULL || cfg->timeout_ms <= 0) {
49 ESP_LOGE(TAG, "Configuracion OTA invalida");
50 return ESP_ERR_INVALID_ARG;
51 }
52
53 ESP_LOGI(TAG, "Iniciando descarga OTA desde %s", cfg->url);
54
55 esp_http_client_config_t http_cfg = {
56 .url = cfg->url,
57 .cert_pem = cfg->cert_pem,
58 .timeout_ms = cfg->timeout_ms,
59 .keep_alive_enable = true,
60 };
61 esp_https_ota_config_t ota_cfg = { .http_config = &http_cfg };
62 esp_https_ota_handle_t handle = NULL;
63
64 esp_err_t err = esp_https_ota_begin(&ota_cfg, &handle);
65 if (err != ESP_OK) {
66 ESP_LOGE(TAG, "esp_https_ota_begin fallo: %s", esp_err_to_name(err));
67 return err;
68 }
69
70 /* Verificacion previa: descriptor de la imagen y version */
71 esp_app_desc_t new_desc;
72 err = esp_https_ota_get_img_desc(handle, &new_desc);
73 if (err != ESP_OK) {
74 ESP_LOGE(TAG, "No se pudo leer el descriptor de la imagen: %s", esp_err_to_name(err));
75 goto abort_ota;
76 }
77 ESP_LOGI(TAG, "Firmware remoto: proyecto '%s', version %s", new_desc.project_name, new_desc.version);
78
79 esp_app_desc_t run_desc;
80 if (cfg->skip_same_version &&
81 esp_ota_get_partition_description(esp_ota_get_running_partition(), &run_desc) == ESP_OK &&
82 strncmp(new_desc.version, run_desc.version, sizeof(new_desc.version)) == 0) {
83 ESP_LOGW(TAG, "La version %s ya esta instalada; no se actualiza", run_desc.version);
84 err = ESP_ERR_INVALID_VERSION;
85 goto abort_ota;
86 }
87
88 /* Descarga y escritura en la particion inactiva */
89 int last_step = -1;
90 while ((err = esp_https_ota_perform(handle)) == ESP_ERR_HTTPS_OTA_IN_PROGRESS) {
91 int total = esp_https_ota_get_image_size(handle);
92 int read = esp_https_ota_get_image_len_read(handle);
93 if (total > 0) {
94 int pct = (int)(((int64_t)read * 100) / total);
95 if (pct / OTA_PROGRESS_STEP != last_step) {
96 last_step = pct / OTA_PROGRESS_STEP;
97 ESP_LOGI(TAG, "Progreso: %d%% (%d/%d bytes)", pct, read, total);
98 }
99 }
100 }
101 if (err != ESP_OK) {
102 ESP_LOGE(TAG, "Descarga fallida: %s", esp_err_to_name(err));
103 goto abort_ota;
104 }
105
106 if (!esp_https_ota_is_complete_data_received(handle)) {
107 ESP_LOGE(TAG, "Descarga incompleta");
108 err = ESP_ERR_INVALID_SIZE;
109 goto abort_ota;
110 }
111
112 /* finish valida la imagen (magic, checksum, hash) y fija la particion de arranque */
113 err = esp_https_ota_finish(handle);
114 if (err != ESP_OK) {
115 if (err == ESP_ERR_OTA_VALIDATE_FAILED) {
116 ESP_LOGE(TAG, "La imagen descargada no supero la validacion");
117 } else {
118 ESP_LOGE(TAG, "esp_https_ota_finish fallo: %s", esp_err_to_name(err));
119 }
120 return err; /* el handle ya fue liberado por finish */
121 }
122
123 ESP_LOGI(TAG, "Firmware verificado e instalado correctamente");
124 return ESP_OK;
125
126abort_ota:
127 esp_https_ota_abort(handle);
128 return err;
129}

153{
154 if (cfg == NULL || cfg->url == NULL || cfg->cert_pem == NULL || cfg->timeout_ms <= 0) {

157 if (s_task_running) {
158 return ESP_ERR_INVALID_STATE;
159 }
160
161 s_task_cfg = *cfg;
162 s_task_running = true;
163 if (xTaskCreate(ota_task, "ota_task", OTA_TASK_STACK_SIZE, NULL, OTA_TASK_PRIORITY, NULL) != pdPASS) {
164 s_task_running = false;

166 }
